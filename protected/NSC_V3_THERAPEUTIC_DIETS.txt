-- Patient discharge support (safe additive change)
alter table public.patients add column if not exists discharged_at timestamptz;

create table if not exists public.therapeutic_diets (
  id uuid primary key default gen_random_uuid(),
  owner_id uuid not null references auth.users(id) on delete cascade,
  name text not null,
  description text,
  is_public boolean not null default false,
  meal_plan jsonb not null default '{"days":[]}'::jsonb,
  created_at timestamptz not null default now(),
  updated_at timestamptz not null default now()
);

create index if not exists therapeutic_diets_owner_idx on public.therapeutic_diets(owner_id);
create index if not exists therapeutic_diets_public_idx on public.therapeutic_diets(is_public);

alter table public.therapeutic_diets enable row level security;

drop policy if exists "therapeutic_diets_select" on public.therapeutic_diets;
create policy "therapeutic_diets_select" on public.therapeutic_diets
for select to authenticated
using (is_public = true or owner_id = auth.uid());

drop policy if exists "therapeutic_diets_insert" on public.therapeutic_diets;
create policy "therapeutic_diets_insert" on public.therapeutic_diets
for insert to authenticated
with check (owner_id = auth.uid());

drop policy if exists "therapeutic_diets_update" on public.therapeutic_diets;
create policy "therapeutic_diets_update" on public.therapeutic_diets
for update to authenticated
using (owner_id = auth.uid())
with check (owner_id = auth.uid());

drop policy if exists "therapeutic_diets_delete" on public.therapeutic_diets;
create policy "therapeutic_diets_delete" on public.therapeutic_diets
for delete to authenticated
using (owner_id = auth.uid());

create or replace function public.set_therapeutic_diets_updated_at()
returns trigger language plpgsql as $$
begin
  new.updated_at = now();
  return new;
end;
$$;

drop trigger if exists therapeutic_diets_updated_at on public.therapeutic_diets;
create trigger therapeutic_diets_updated_at
before update on public.therapeutic_diets
for each row execute function public.set_therapeutic_diets_updated_at();
